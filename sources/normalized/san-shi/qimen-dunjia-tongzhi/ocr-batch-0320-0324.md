```python
# 实测文字块 + 竖排指纹（可复现；324 页块宽异常可据此复测）
from PIL import Image
import numpy as np
def fund(v, minlag=8, maxlag=260, thr=0.10):
    v = v - v.mean(); ac = np.correlate(v, v, "full")[len(v)-1:]; ac = ac/ac[0]
    c = [(float(ac[l]), l) for l in range(minlag, min(maxlag, len(ac)-1))
         if ac[l] > ac[l-1] and ac[l] >= ac[l+1] and ac[l] > thr]
    return max(c)[1] if c else None
for p in (321, 324):
    im = Image.open(f'/tmp/qm-tail2/render400/pg-{p}.png').convert('L')
    a = np.asarray(im); ink = a < 140
    xs = np.where(ink.sum(0) > 60)[0]; ys = np.where(ink.sum(1) > 60)[0]
    print(p, 'block', int(xs[-1]-xs[0]), int(ys[-1]-ys[0]),
          'Xcol', fund(ink.sum(0)), 'Ychar', fund(ink.sum(1)))
# -> 321 block 1411 1455 Xcol 228 Ychar 67  => UPRIGHT
# -> 324 block  858 1343 Xcol 226 Ychar 67  => UPRIGHT (块宽异常，照实登记)
```
